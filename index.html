-- ============================================================
-- LAB INFORMATION SYSTEM (LIS) - DATABASE SCRIPT
-- Engine: MySQL 8.0+ (InnoDB)
-- ============================================================

DROP DATABASE IF EXISTS lab_info_system;
CREATE DATABASE lab_info_system CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE lab_info_system;

-- ------------------------------------------------------------
-- 1. PATIENT
-- ------------------------------------------------------------
CREATE TABLE patient (
    patient_id      INT AUTO_INCREMENT PRIMARY KEY,
    first_name      VARCHAR(50)  NOT NULL,
    last_name       VARCHAR(50)  NOT NULL,
    date_of_birth   DATE         NOT NULL,
    gender          ENUM('Male','Female','Other') NOT NULL,
    contact_number  VARCHAR(20),
    address         VARCHAR(255),
    email           VARCHAR(100),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- 2. DOCTOR
-- ------------------------------------------------------------
CREATE TABLE doctor (
    doctor_id       INT AUTO_INCREMENT PRIMARY KEY,
    first_name      VARCHAR(50)  NOT NULL,
    last_name       VARCHAR(50)  NOT NULL,
    specialization  VARCHAR(100),
    contact_number  VARCHAR(20),
    email           VARCHAR(100)
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- 3. STAFF (lab technicians, pathologists, admins, phlebotomists)
-- ------------------------------------------------------------
CREATE TABLE staff (
    staff_id        INT AUTO_INCREMENT PRIMARY KEY,
    first_name      VARCHAR(50)  NOT NULL,
    last_name       VARCHAR(50)  NOT NULL,
    role            ENUM('Medtech','Pathologist','Admin','Phlebotomist') NOT NULL,
    contact_number  VARCHAR(20),
    username        VARCHAR(50)  NOT NULL UNIQUE,
    password_hash   VARCHAR(255) NOT NULL
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- 4. TEST_CATALOG (master list of tests the lab offers)
-- ------------------------------------------------------------
CREATE TABLE test_catalog (
    test_id          INT AUTO_INCREMENT PRIMARY KEY,
    test_name        VARCHAR(100) NOT NULL,
    test_category    VARCHAR(50),
    specimen_type    VARCHAR(50),
    price            DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    turnaround_hours INT,
    normal_range     VARCHAR(100),
    unit             VARCHAR(20)
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- 5. LAB_ORDER (a physician's request for a patient)
-- ------------------------------------------------------------
CREATE TABLE lab_order (
    order_id     INT AUTO_INCREMENT PRIMARY KEY,
    patient_id   INT NOT NULL,
    doctor_id    INT NOT NULL,
    order_date   DATETIME DEFAULT CURRENT_TIMESTAMP,
    status       ENUM('Pending','In Progress','Completed','Cancelled') DEFAULT 'Pending',
    priority     ENUM('Routine','Urgent','STAT') DEFAULT 'Routine',
    CONSTRAINT fk_order_patient FOREIGN KEY (patient_id) REFERENCES patient(patient_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_order_doctor FOREIGN KEY (doctor_id) REFERENCES doctor(doctor_id)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- 6. ORDER_TEST (line items: which tests belong to which order)
-- ------------------------------------------------------------
CREATE TABLE order_test (
    order_test_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id      INT NOT NULL,
    test_id       INT NOT NULL,
    status        ENUM('Pending','Collected','Resulted','Verified') DEFAULT 'Pending',
    CONSTRAINT fk_ordertest_order FOREIGN KEY (order_id) REFERENCES lab_order(order_id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_ordertest_test FOREIGN KEY (test_id) REFERENCES test_catalog(test_id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT uq_order_test UNIQUE (order_id, test_id)
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- 7. SPECIMEN (sample collected for one order_test line item)
-- ------------------------------------------------------------
CREATE TABLE specimen (
    specimen_id      INT AUTO_INCREMENT PRIMARY KEY,
    order_test_id    INT NOT NULL UNIQUE,
    specimen_type    VARCHAR(50),
    collection_date  DATETIME,
    collected_by     INT,
    status           ENUM('Collected','Received','Rejected') DEFAULT 'Collected',
    CONSTRAINT fk_specimen_ordertest FOREIGN KEY (order_test_id) REFERENCES order_test(order_test_id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_specimen_staff FOREIGN KEY (collected_by) REFERENCES staff(staff_id)
        ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- 8. TEST_RESULT (result for one specimen)
-- ------------------------------------------------------------
CREATE TABLE test_result (
    result_id        INT AUTO_INCREMENT PRIMARY KEY,
    specimen_id       INT NOT NULL UNIQUE,
    result_value      VARCHAR(100),
    unit              VARCHAR(20),
    reference_range   VARCHAR(100),
    flag              ENUM('Normal','High','Low','Critical') DEFAULT 'Normal',
    result_date       DATETIME DEFAULT CURRENT_TIMESTAMP,
    verified_by       INT,
    remarks           VARCHAR(255),
    CONSTRAINT fk_result_specimen FOREIGN KEY (specimen_id) REFERENCES specimen(specimen_id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_result_staff FOREIGN KEY (verified_by) REFERENCES staff(staff_id)
        ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- INDEXES (speed up common lookups beyond PK/FK auto-indexes)
-- ------------------------------------------------------------
CREATE INDEX idx_patient_name       ON patient (last_name, first_name);
CREATE INDEX idx_order_status       ON lab_order (status);
CREATE INDEX idx_order_date         ON lab_order (order_date);
CREATE INDEX idx_ordertest_status   ON order_test (status);
CREATE INDEX idx_result_flag        ON test_result (flag);

LIS-05 / INDEX
05 — Index

Why each part exists

What each section is used for, and why it was included in this package.

01
Cover material

Sets the scope up front — what system this is, what engine it targets, and what's included — so the package can be handed off without a verbal walkthrough.

02
ER diagram

Created first because every other artifact derives from it. It fixes the entities and cardinalities before a single CREATE TABLE is written, catching design mistakes while they're still cheap to change.

03
Schema table

Used as the bridge between the diagram and the code — a plain-language contract listing every column, type, and key so a reviewer can sanity-check the design without reading SQL.

04
SQL script

The buildable artifact. Everything above exists to justify what's in this file — it's the only piece that actually creates the database.

05
This index

Included because a design package without a rationale gets questioned later — this records why each part exists, not just what it contains.
